package httpapi_test

import (
 "bytes"
 "net/http"
 "net/http/httptest"
 "path/filepath"
 "routeweaver/internal/httpapi"
 "routeweaver/internal/render"
 "routeweaver/internal/store"
 "strings"
 "testing"
)

func TestIndependentForwardedHeadersCannotGrantMutationOrigin(t *testing.T) {
 s, err := store.Open(filepath.Join(t.TempDir(), "db"), bytes.Repeat([]byte{41}, 32))
 if err != nil { t.Fatal(err) }; defer s.Close()
 for _, mode := range []string{"unconfigured", "wrong-host", "wrong-origin", "null-origin"} {
  origin, host, public := "https://cp.example", "cp.example", "https://cp.example"
  switch mode { case "unconfigured": public = ""; case "wrong-host": host = "evil.example"; case "wrong-origin": origin = "https://evil.example"; case "null-origin": origin = "null" }
  handler := httpapi.NewWithOptions(s, render.Default(), httpapi.Options{PublicOrigin: public})
  req := httptest.NewRequest(http.MethodPost, "http://127.0.0.1/api/auth/login", strings.NewReader(`{}`))
  req.Host = host
  req.Header.Set("Origin", origin)
  req.Header.Set("X-Forwarded-Proto", "https")
  req.Header.Set("X-Forwarded-Host", "cp.example")
  req.Header.Set("Forwarded", "proto=https;host=cp.example")
  got := httptest.NewRecorder(); handler.ServeHTTP(got, req)
  if got.Code != http.StatusForbidden { t.Fatalf("%s accepted forged forwarding proof: %d %s", mode, got.Code, got.Body.String()) }
 }
}
